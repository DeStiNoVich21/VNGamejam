// ── ТЕСТОВЫЙ ЗАПУСК: СИСТЕМА ПЕРСОНАЖЕЙ ──────────────────────────
NextDay() // Переход к техническому дню
SetLayerMedia(background bridge_view)
playambience(Tranquil)

Narrator "(Запуск протокола тестирования экипажа...)"

// 1. ТЕСТ СПАВНА И ПОЗИЦИИ
// Создаем Джонатана (из твоего списка) и ставим его по центру
CreateCharacter(Jonathan)
Jonathan.SetPosition(0.5 0.2)
Jonathan.Show()

Jonathan "Капитан, проверка связи. Я в центре экрана. {spd=0.5}Слышите меня?{/spd}"
Jonathan [0:gg1] "Привет!"
Jonathan [0:gg2] "Я удивлён."
Jonathan [0:gg6] "Дефолт."
// 2. ТЕСТ ЭФФЕКТОВ В ДИАЛОГЕ
// Проверяем, как на нем смотрятся разные теги
Jonathan "{shake=1}ОЙ!{/shake} Кажется, палуба дрожит. Или это у меня процессор барахлит?"
Jonathan "{rainbow=1}Смотрите, я поймал сигнал из туманности!{/rainbow} Цвета... они текут."

// 3. ТЕСТ ВЗАИМОДЕЙСТВИЯ (choice)
choice "Как ответить Джонатану?"
{
    - "Джонатан, встань слева"
        Jonathan.SetPosition(0.2 0.2)
        Jonathan "Принято. Смещаюсь в левый сектор. Тут обзор на звезды лучше."
    
    - "Джонатан, встань справа"
        Jonathan.SetPosition(0.8 0.2)
        Jonathan "Есть! Правый борт под контролем. Жду дальнейших указаний."
}

// 4. ТЕСТ ГРУППОВОГО СПАВНА
// Проверим, не "ломается" ли сцена при добавлении других из твоего списка
CreateCharacter(Emily) // Та самая, у которой в Alias было "Emly"
Emily.SetPosition(0.4 0.2)
Emily.Show()

Emily "{wave=0.4}Привет!{/wave} Я тоже здесь. Проверяю Font Scale и скорость текста."

// 5. ТЕСТ ИСЧЕЗНОВЕНИЯ
Narrator "(Завершение визуального теста...)"

Jonathan.Hide()
Emily.Hide()

ClearLayerMedia(background)
stopambience()

Narrator "(Все персонажи успешно выгружены. Ошибок в слоях не обнаружено.)"
PauseConversation()

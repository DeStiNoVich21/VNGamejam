if ($sync_dominion >= $sync_zenith && $sync_dominion >= $sync_stigma)
{
    set(route_dominion = true)
    set(current_route = "Dominion")
    raise_sync(dominion 1)
}
else
{
    if ($sync_zenith >= $sync_stigma)
    {
        set(route_zenith = true)
        set(current_route = "Zenith")
        raise_sync(zenith 1)
    }
    else
    {
        set(route_stigma = true)
        set(current_route = "Stigma")
        raise_sync(stigma 1)
    }
}

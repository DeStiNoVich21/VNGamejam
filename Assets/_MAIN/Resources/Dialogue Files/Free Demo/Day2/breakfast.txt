// --- СЦЕНА: УТРО В КАФЕ ---
SetLayerMedia(background cafe_morning)
playambience(CafeMorning)
playsong(DetectiveTheme)

$playerName = "Джон"

// --- ПЕРСОНАЖИ ---
CreateCharacter(John), John.SetPosition(-0.02 2.5)
CreateCharacter(Cecilia), Cecilia.SetPosition(1.02 2.5)
CreateCharacter(Lance), Lance.SetPosition(1.02 2.5)

// --- ВСТУПЛЕНИЕ ---
John.Show()
John "(Перед отправкой в клуб надо зайти позавтракать.)"

Cecilia.Show()
Cecilia "Мистер Джонатан, добро пожаловать!"

John "Доброе утро, Сесилия."

Cecilia "Чего желаете? Вам как всегда?"

John "Вы меня слишком хорошо знаете."

Cecilia "Поняла. Подождите пока здесь."
Cecilia.Hide()

// --- ОЖИДАНИЕ ---
John "(Пока Сесилия готовила и обслуживала других гостей, я вдумчиво смотрел на часы.){pause=1.0} (Он всегда приходит в одно и тоже время.)"

// --- ПОЯВЛЕНИЕ ЛЭНСА ---
PlaySFX(DoorOpen)
Lance.Show(), Lance.FaceLeft(-i true)

John "(Дверь сзади меня распахнулась и вошел мой {c=#87CEEB}белобрысый{/c} друг.)"

John "Я не перестаю удивляться твоей пунктуальности. Ты там в случаем не залипаешь на время перед входом?"

Lance "{wave=0.5}Ну что тут сказать{/wave}, пунктуальность — залог успеха, Джонни. Я всегда прихожу в нужный момент."

John "Ну да, ну да. Как скажешь."

// --- РАЗГОВОР О ДЕЛЕ ---
Lance "Как продвигается дело? Что-то выяснил?"

John "Расследование о вчерашней бойне привел меня к дилеру с низов.{pause=0.5} Пришлось его напрячь, чтобы он помог мне."

Lance "И что он ответил?"

John "Вчера ночью он связался со мной, чтобы что-то важное обсудить.{c} После еды собираюсь сразу выдвигаться."

John "(На лице Лэнса появилась ухмылка. Сейчас он скажет что-то глупое.)"

Lance "{i}Понятно{/i}. Детектив и синдикат работают вместе.{pause=0.5} Таки напишу в отчете."

John "Не бери в голову. Работа есть работа."

// --- ВОЗВРАТ СЕСИЛИИ ---
Cecilia.Show()

PlaySFX(PlateSet)
Cecilia "Ваш заказ! Приятного аппетита."

John "Спасибо вам."

John "(Позже и Лэнс заказал себе завтрак с капучино, и мы обсуждали свои будни.)"

// --- ЗВОНОК ---
PlaySFX(PhoneRing)

John "(Внезапно, мой телефон зазвенел. Номер отмечен как {c=#FF0000}неизвестный{/c}.)"

Lance "Я так понял это он?"

John "Видимо."

// --- ВЫБОР ИГРОКА ---
choice "Ответить на звонок?"
{
    - Ответить сразу
        $john.attitude += 1
        John "Слушаю вас."

    - Немного подождать
        wait(1.5)
        $john.attitude -= 1
        John "Да... слушаю."
}

// --- ДИАЛОГ С ДИЛАНОМ ---
CreateCharacter(Dylan)

Dylan "{spd=0.8}Это детектив Джонатан?{/spd}"

John "Да, в чем дело?"

Dylan "Это друг дилера. Он попросил меня заранее позвонить вам на случай если что-то произойдет."

John "{shake=0.5}Что там у вас?{/shake}"

Dylan "{c=#FF0000}Его убили…{/c}{pause=1.0} Мы нашли его мертвым в его комнате."

PlaySFX(TensionHit)
stopambience()
playambience(Tension)

Dylan "Сюда уже приехали копы. Это не вы их в случае созвали?! У нас же был уговор!"

John "Не может быть. Я только сейчас об этом узнаю. Никаких сведений мне еще не поступало."

Dylan "Вот как значит... В любом случае поторопитесь и уладьте дело здесь быстро."

John "Уже еду."

// --- ЗАВЕРШЕНИЕ СЦЕНЫ ---
Lance "Что там у тебя?"

John "ЧП. Расскажу позже."

Lance "Хорошо. Удачи с делом."

John "(Я оставил деньги на столе и сразу вышел из кофейни.)"

PlaySFX(ChairMove)
PlaySFX(DoorOpen)

John.Hide()
Lance.Hide()
Cecilia.Hide()
